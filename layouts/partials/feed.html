{{/* Shared feed renderer: item list + infinite scroll. Expects a dict: - "page": the list page context (owner of the paginator) - "pages": the slice of pages to paginate - "item": optional partial name to render each page (defaults to "feed-item.html") Bookmark pages always render through the compact "bookmark-item.html". Renders 10 items per page. */}} {{ $page := .page }} {{ $item := .item | default "feed-item.html" }} {{ $paginator := $page.Paginate .pages 10 }}
{{ range $paginator.Pages }} {{ if eq .Type "bookmark" }} {{ partial "bookmark-item.html" . }} {{ else }} {{ partial $item . }} {{ end }} {{ end }}
{{ with $paginator.Next }} Load more {{ end }}
{{ partial "infinite-feed.html" $page }}
